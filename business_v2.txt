        .btn-group { margin-top: 15px; display: flex; gap: 10px; flex-wrap: wrap; }
        
        .nav-btn { color: white; padding: 12px 20px; text-decoration: none; font-weight: bold; border-radius: 4px; display: inline-block; font-size: 14px; text-align: center; }

        button.danger:hover { background: #f87171; }

        .news-box { background: #ffea00; color: #000; padding: 12px; border-radius: 4px; font-weight: bold; margin-bottom: 20px; border-left: 5px solid #ffaa00; font-size: 15px; }

    <script>
        let params = new URLSearchParams(window.location.search);
        let cash = params.get('cash') ? parseInt(params.get('cash')) : 2000000;
        let loan = params.get('loan') ? parseInt(params.get('loan')) : 0;
        let deposit = params.get('deposit') ? parseInt(params.get('deposit')) : 0;
        
        // 在庫データの引き継ぎ（4大事業に対応）
        let stockPhone = params.get('sPhone') ? parseInt(params.get('sPhone')) : 0;
        let stockCar = params.get('sCar') ? parseInt(params.get('sCar')) : 0;
        let stockRobot = params.get('sRobot') ? parseInt(params.get('sRobot')) : 0; // AIロボット
        let stockRocket = params.get('sRocket') ? parseInt(params.get('sRocket')) : 0; // ロケット
        
        let staff = params.get('staff') ? parseInt(params.get('staff')) : 1;
        let term = params.get('term') ? parseInt(params.get('term')) : 1;
        
        let newsText = params.get('news') ? params.get('news') : "安定した経済環境です。さらなる事業拡大を目指しましょう！";
        let priceMultiplier = params.get('mult') ? parseFloat(params.get('mult')) : 1.0;

        function updateUI() {
            document.getElementById("lbl-term").innerText = term;
            document.getElementById("lbl-cash").innerText = cash.toLocaleString();
            document.getElementById("lbl-loan").innerText = loan.toLocaleString();
            document.getElementById("lbl-deposit").innerText = deposit.toLocaleString();
            document.getElementById("lbl-staff").innerText = staff;
            
            document.getElementById("lbl-stock-phone").innerText = stockPhone;
            document.getElementById("lbl-stock-car").innerText = stockCar;
            document.getElementById("lbl-stock-robot").innerText = stockRobot;
            document.getElementById("lbl-stock-rocket").innerText = stockRocket;
            
            document.getElementById("news-display").innerText = newsText;

            // 他の部屋へのパスに必要な全データを上書き
            let dataQuery = `?cash=${cash}&loan=${loan}&deposit=${deposit}&sPhone=${stockPhone}&sCar=${stockCar}&sRobot=${stockRobot}&sRocket=${stockRocket}&staff=${staff}&term=${term}&news=${encodeURIComponent(newsText)}&mult=${priceMultiplier}`;
            
            document.getElementById("lnk-bank").href = "bank_v2.html" + dataQuery;
            document.getElementById("lnk-business").href = "business_v2.html" + dataQuery;
            document.getElementById("lnk-hr").href = "hr_v2.html" + dataQuery;
        }

        function logAction(text) {
            let logBox = document.getElementById("log");
            if (logBox) { logBox.innerHTML += "<br>" + text; logBox.scrollTop = logBox.scrollHeight; }
        }

        // ⏱ 決算を行い、時間を翌月に進める（4大事業の販売対応）
        function nextMonth() {
            logAction(`--- 【${term}ヶ月目 月次決算報告】 ---`);
            
            // 各製品の自動売却処理（在庫があれば売れる）
            if (stockPhone > 0) { 
                let p = Math.round(70000 * priceMultiplier); stockPhone--; cash += p; 
                logAction(`[営業] スマホが売却されました。売上: +${p.toLocaleString()}円`); 
            }
            if (stockCar > 0) { 
                let p = Math.round(2500000 * priceMultiplier); stockCar--; cash += p; 
                logAction(`[営業] 電気自動車(EV)が売却されました。売上: +${p.toLocaleString()}円`); 
            }
            if (stockRobot > 0) { 
                let p = Math.round(10000000 * priceMultiplier); stockRobot--; cash += p; 
                logAction(`[営業] 🤖人型AIロボットが工場へ納品・売却されました。売上: +${p.toLocaleString()}円`); 
            }
            if (stockRocket > 0) { 
                let p = Math.round(45000000 * priceMultiplier); stockRocket--; cash += p; 
                logAction(`[営業] 🚀次世代ロケットが宇宙機関へ売却されました！売上: +${p.toLocaleString()}円`); 
            }

            // 維持費（家賃15万 ＋ 社員の給料1人20万）
            let salaryCost = staff * 200000;
            let officeRent = 150000;
            cash -= (salaryCost + officeRent);
            logAction(`[財務] 固定経費引落：オフィス家賃 ${officeRent.toLocaleString()}円 / 人件費 ${salaryCost.toLocaleString()}円`);

            // 銀行利息
            cash = cash - Math.round(loan * 0.005) + Math.round(deposit * 0.002);

            if (cash < 0) { 
                alert("【資金ショート】手元金がマイナスになり破産しました。"); 
                window.location.href = "ceo_v2.html"; 
                return; 
            }

            // 📰 翌月のランダム経済ニュース
            let rand = Math.random();
            if (rand < 0.20) { 
                newsText = "【経済ニュース：世界的なテクノロジー大バブル到来！】 AIロボットと宇宙ロケットの需要が爆発！売価が【1.5倍】に高騰中！"; 
                priceMultiplier = 1.5; 
            } else if (rand < 0.40) { 
                newsText = "【経済ニュース：世界同時不況の波】 市場が大混乱。全製品の市場引き取り価格が【半分(0.5倍)】に大暴暴落！"; 
                priceMultiplier = 0.5; 
            } else if (rand < 0.55) {
                newsText = "【経済ニュース：ハイテク減税政策が決定】 デジタルガジェットが大人気！スマホと自動車の売価が【1.2倍】にアップ！";
                priceMultiplier = 1.2;
            } else { 
                newsText = "【経済ニュース：市場は極めて安定】 特記すべき経済変動はありません。安定した取引が続いています。"; 
                priceMultiplier = 1.0; 
            }

            term++;
            updateUI();
        }
    </script>
